const samples = [
 { text: 'こんにちは', file: '125aeadf27b0459b8760.mp3' },
 { text: '先生', file: 'cd98e43da31bb1b90dc1.mp3' },
 { text: '愛情', file: 'n3/6da765bf0fca68170b8f.mp3' },
];
const audio = document.getElementById('audio'), text = document.getElementById('text');
const status = document.getElementById('status'), note = document.getElementById('device-note');
let relayOrigin = '', sequence = 0, controller, objectURL, context, sourceNode;
const clips = new Map();
function say(message, error = false) { status.textContent = message; status.className = 'status' + (error ? ' error' : ''); }
function stop() {
 sequence++; controller?.abort(); controller = undefined;
 if (sourceNode) { sourceNode.onended = null; try { sourceNode.stop(); } catch {} sourceNode.disconnect(); sourceNode = undefined; }
 audio.pause(); audio.removeAttribute('src'); audio.load();
 if (objectURL) URL.revokeObjectURL(objectURL); objectURL = undefined;
 window.speechSynthesis?.cancel();
}
audio.addEventListener('error', () => say('音频未能加载，请确认网站网络连接后重试。', true));
audio.addEventListener('ended', () => say('播放完毕，可以选择其他示例。'));
document.querySelectorAll('[data-sample]').forEach(button => button.addEventListener('click', () => {
 stop(); const sample = samples[Number(button.dataset.sample)]; audio.src = '../audio/japanese/' + sample.file;
 say('正在播放内置音频 · ' + sample.text);
 audio.play().catch(() => say('浏览器没有开始播放，请点击播放器的播放键。', true));
}));
document.getElementById('stop').addEventListener('click', () => { stop(); say('已停止。'); });
document.getElementById('play-device').addEventListener('click', () => {
 stop(); note.textContent = '';
 if (!text.value.trim()) { text.focus(); return; }
 const synth = window.speechSynthesis;
 const voice = synth?.getVoices().find(item => /^ja(?:-|_)/i.test(item.lang));
 if (!voice || typeof SpeechSynthesisUtterance === 'undefined') { note.textContent = '本浏览器暂未提供日语声音，可先试听上方内置音频。'; return; }
 const active = sequence, utterance = new SpeechSynthesisUtterance(text.value.trim());
 utterance.lang = 'ja-JP'; utterance.voice = voice; utterance.rate = .85;
 let started = false;
 const timeout = setTimeout(() => { if (active === sequence && !started) { synth.cancel(); note.textContent = '设备语音未能开始播放，可先使用内置音频。'; } }, 5000);
 utterance.onstart = () => { started = true; clearTimeout(timeout); note.textContent = '正在播放设备日语语音。'; };
 utterance.onend = () => { clearTimeout(timeout); if (active === sequence) note.textContent = '设备语音播放完毕。'; };
 utterance.onerror = () => { clearTimeout(timeout); if (active === sequence) note.textContent = '设备未能播放这段日语，可先使用内置音频。'; };
 synth.speak(utterance);
});
window.speechSynthesis?.getVoices();
fetch('./config.json', { cache: 'no-store' }).then(response => response.ok ? response.json() : Promise.reject()).then(config => {
 if (!config.relayOrigin) return;
 const url = new URL(config.relayOrigin);
 if (url.protocol !== 'https:' || url.username || url.password || url.hostname.endsWith('.chatgpt.site')) return;
 relayOrigin = url.origin;
 document.getElementById('play-online').disabled = false;
 document.getElementById('relay-badge').textContent = '已接入'; document.getElementById('relay-badge').className = 'badge';
 document.getElementById('relay-note').textContent = '已接入新的实时语音服务。首次播放需要联网；若网络不通，上方内置音频仍可试听。';
}).catch(() => {});
document.getElementById('play-online').addEventListener('click', async () => {
 if (!relayOrigin || !text.value.trim()) return;
 stop(); const active = sequence;
 // Unlock Web Audio inside the tap; loading and decoding happen afterward.
 let ready = Promise.resolve(false);
 try { const C = window.AudioContext || window.webkitAudioContext; if (C) { context ??= new C(); ready = context.state === 'running' ? Promise.resolve(true) : context.resume().then(() => context.state === 'running', () => false); } } catch {}
 const voice = document.getElementById('voice').value;
 const url = relayOrigin + '/speech?' + new URLSearchParams({ voice, text: text.value.trim() });
 controller = new AbortController(); const signal = controller.signal;
 const timer = setTimeout(() => { if (active === sequence) controller?.abort(); }, 15000);
 say('正在获取 ' + (voice.includes('Nanami') ? 'Nanami' : 'Keita') + ' 语音…');
 try {
  let blob = clips.get(url);
  if (!blob) {
   const response = await fetch(url, { signal, credentials: 'omit', cache: 'no-store' });
   if (!response.ok || !(response.headers.get('Content-Type') || '').startsWith('audio/')) throw new Error('unavailable');
   blob = await response.blob(); if (blob.size < 256) throw new Error('empty');
   clips.set(url, blob); while (clips.size > 8) clips.delete(clips.keys().next().value);
  }
  if (active !== sequence || signal.aborted) return;
  if (await ready) {
   const buffer = await context.decodeAudioData(await blob.arrayBuffer());
   if (active !== sequence || signal.aborted) return;
   sourceNode = context.createBufferSource(); sourceNode.buffer = buffer; sourceNode.connect(context.destination);
   sourceNode.onended = () => { if (active === sequence) say('播放完毕。'); }; sourceNode.start();
  } else { objectURL = URL.createObjectURL(blob); audio.src = objectURL; await audio.play(); }
  say('正在播放微软实时语音。');
 } catch (error) { if (active === sequence) say(error?.name === 'NotAllowedError' ? '请再次点击播放，或点击下方播放器的播放键。' : '实时语音未能连接或播放，可先试听上方内置音频。', true); }
 finally { clearTimeout(timer); }
});
window.addEventListener('pagehide', () => { stop(); void context?.close(); clips.clear(); });
